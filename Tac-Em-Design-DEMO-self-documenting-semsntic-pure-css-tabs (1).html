Using Radio Buttons for State

HTML for this Demo

<nav role="tablist" aria-label="Tabbed Content Sections">

  <!-- Hidden Radio Inputs Manage State -->
  <label for="sec1-tgl"
     role="tab"
     id="sec1-lbl"
     aria-controls="sec1">
  HTML
  <input type="radio"
     name="tabs-group"
     id="sec1-tgl"
     hidden checked>
  </label>
  <section id="sec1"
     role="tabpanel"
     aria-labelledby="sec1-lbl">
     <h3>HTML for this Demo</h3>
     <pre><code>... HTML code goes here ...</code></pre>
  </section>
  <!-- END Section 1 -->

  <label for="sec2-tgl"
     role="tab"
     aria-controls="sec2">
  CSS
  <input type="radio"
     name="tabs-group"
     id="sec2-tgl" hidden>
  </label>

  <section id="sec2"
     role="tabpanel"
     aria-labelledby="sec2-lbl">
     <h3>CSS for this Demo</h3>
     <pre><code>... CSS code goes here ...</code></pre>
  </section>

</nav>

   flex-direction: column;
}

/* --- Tab Control Styling --- */
[role='tab'] {
   /* ... tab styles ... */
   margin-bottom: -3px; /* Overlap */
   z-index: 1;
}

/* --- CORE TacEm Logic & Active State Styling --- */
[role='tab']:has(input:checked) + [role='tabpanel'] {
   visibility: visible;
   z-index: 2;
}
[role='tab']:has(:checked) {
   /* ... active tab styles ... */
   background-color: var(--pnl);
   border-bottom-color: var(--pnl);
   z-index: 3;
}

/* Accessibility: Focus styles */
[role='tab']:focus-visible {
   /* ... focus styles ... */
   z-index: 4;
}
